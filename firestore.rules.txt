rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Orders: anyone can create a new order (so customers can check
    // out without logging in), but only a signed-in admin can read,
    // update, or delete them. Nobody can overwrite an existing order.
    match /orders/{orderId} {
      allow create: if true;
      allow read, update, delete: if request.auth != null;
    }

    // Products: anyone can read the catalog (so the storefront can
    // show it to shoppers who aren't logged in), but only a signed-in
    // admin can add, edit, or delete products.
    match /products/{productId} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // Settings (e.g. which checkout fields to collect): anyone can
    // read so the storefront can build the checkout form, but only a
    // signed-in admin can change the configuration.
    match /settings/{settingId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
  }
}
