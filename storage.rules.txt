rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {

    // Product photos: anyone can view them (customers need to see
    // your product images), but only a signed-in admin can upload
    // or replace them from the admin dashboard.
    match /product-images/{fileName} {
      allow read: if true;
      allow write: if request.auth != null
                   && request.resource.size < 10 * 1024 * 1024
                   && request.resource.contentType.matches('image/.*');
    }
  }
}
